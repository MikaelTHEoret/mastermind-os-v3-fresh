### 12.8 Strategic Overlay Control System

The Strategic Overlay Control System provides a unified interface for managing and visualizing trading strategies, risk parameters, and signal processing across the Mastermind System.

```python
class StrategicOverlayControl:
    """
    Strategic overlay control system for managing high-level system parameters
    
    Features:
    - Strategy selection and configuration
    - Risk mode management
    - Signal input control
    - Live simulation and monitoring
    - Gradient-based control optimization
    """
    
    def __init__(self, data_manager, glyph_router, strategy_ontology):
        self.data_manager = data_manager
        self.glyph_router = glyph_router
        self.strategy_ontology = strategy_ontology
        self.vector_db = self._initialize_vector_db()
        
        # Initialize control parameters
        self.control_state = {
            'selected_strategy': None,
            'risk_mode': 'medium',  # low, medium, high
            'forecast_horizon': 10, # steps
            'external_feeds_enabled': True,
            'latent_factors_enabled': True,
            'control_cost_estimator': 'auto',
            'simulation_active': False,
            'regret_bound_type': 'GPC'  # GPC or GRC
        }
        
        # Initialize performance metrics
        self.metrics = {
            'regret_curve': [],
            'benchmark_delta': [],
            'last_update': None
        }
        
        # Subscribe to glyphs
        self._setup_glyph_subscriptions()
    
    def _initialize_vector_db(self):
        """Initialize vector database for strategy and signal storage"""
        try:
            import chromadb
            client = chromadb.Client()
            
            # Create collections
            verse_protocols = client.create_collection("verse_protocols")
            event_log = client.create_collection("event_log")
            
            return {
                'client': client,
                'collections': {
                    'verse_protocols': verse_protocols,
                    'event_log': event_log
                }
            }
        except ImportError:
            logging.warning("ChromaDB not available, using dictionary-based storage")
            return {
                'client': None,
                'collections': {
                    'verse_protocols': {},
                    'event_log': {}
                }
            }
    
    def _setup_glyph_subscriptions(self):
        """Setup subscriptions for relevant glyphs"""
        if self.glyph_router:
            # Subscribe to strategy-related glyphs
            self.glyph_router.subscribe(
                'STRATEGY_*',
                self._handle_strategy_glyph,
                priority=10
            )
            
            # Subscribe to signal-related glyphs
            self.glyph_router.subscribe(
                'SIGNAL_*',
                self._handle_signal_glyph,
                priority=5
            )
    
    def _handle_strategy_glyph(self, event):
        """Handle strategy-related glyphs"""
        glyph = event['glyph']
        payload = event['payload']
        
        if glyph == 'STRATEGY_ACTIVATED':
            # Store strategy in vector DB
            self._store_strategy_in_vectordb(payload)
            
            # Update control state
            if self.control_state['selected_strategy'] is None:
                self.select_strategy(payload['strategy_id'])
        
        elif glyph == 'STRATEGY_PERFORMANCE_UPDATE':
            # Update metrics
            self._update_performance_metrics(payload)
            
            # Store signal in vector DB
            self._store_signal_in_vectordb({
                'id': f"perf-{uuid.uuid4().hex[:8]}",
                'document': f"Performance Update: {payload['strategy_id']}",
                'metadata': {
                    'timestamp': datetime.now().isoformat(),
                    'strategy_id': payload['strategy_id'],
                    'performance': payload['performance'],
                    'type': 'performance_update'
                }
            })
    
    def _handle_signal_glyph(self, event):
        """Handle signal-related glyphs"""
        glyph = event['glyph']
        payload = event['payload']
        
        if glyph in ['SIGNAL_REGISTERED', 'SIGNAL_ACTED_UPON', 'SIGNAL_EXPIRED']:
            # Store signal in vector DB
            self._store_signal_in_vectordb({
                'id': f"signal-{uuid.uuid4().hex[:8]}",
                'document': f"Signal {glyph.split('_')[1]}: {payload.get('type')}",
                'metadata': {
                    'timestamp': datetime.now().isoformat(),
                    'signal_id': payload.get('signal_id'),
                    'signal_type': payload.get('type'),
                    'symbol': payload.get('symbol'),
                    'strength': payload.get('initial_strength', payload.get('strength_at_action', 0)),
                    'glyph': glyph
                }
            })
            
            # Update control metrics if simulation is active
            if self.control_state['simulation_active']:
                self._calculate_regret(payload)
    
    def _store_strategy_in_vectordb(self, strategy_data):
        """Store strategy in vector DB"""
        if self.vector_db['client'] is None:
            # Dictionary-based fallback
            self.vector_db['collections']['verse_protocols'][strategy_data['strategy_id']] = {
                'id': f"strat-{strategy_data['strategy_id']}",
                'document': f"Strategy: {strategy_data.get('name', strategy_data['strategy_id'])}",
                'metadata': {
                    'type': strategy_data.get('type', 'unknown'),
                    'algorithm': strategy_data.get('algorithm', 'unknown'),
                    'regret_bound': self._get_regret_bound(strategy_data.get('type', 'unknown')),
                    'loss_convexity': True,
                    'memory_window': "h",
                    'notes': strategy_data.get('description', '')
                }
            }
        else:
            # ChromaDB storage
            collection = self.vector_db['collections']['verse_protocols']
            collection.add(
                ids=[f"strat-{strategy_data['strategy_id']}"],
                documents=[f"Strategy: {strategy_data.get('name', strategy_data['strategy_id'])}"],
                metadatas=[{
                    'type': strategy_data.get('type', 'unknown'),
                    'algorithm': strategy_data.get('algorithm', 'unknown'),
                    'regret_bound': self._get_regret_bound(strategy_data.get('type', 'unknown')),
                    'loss_convexity': True,
                    'memory_window': "h",
                    'notes': strategy_data.get('description', '')
                }]
            )
    
    def _store_signal_in_vectordb(self, signal_data):
        """Store signal in vector DB"""
        if self.vector_db['client'] is None:
            # Dictionary-based fallback
            self.vector_db['collections']['event_log'][signal_data['id']] = signal_data
        else:
            # ChromaDB storage
            collection = self.vector_db['collections']['event_log']
            collection.add(
                ids=[signal_data['id']],
                documents=[signal_data['document']],
                metadatas=[signal_data['metadata']]
            )
    
    def _get_regret_bound(self, strategy_type):
        """Get appropriate regret bound formula for strategy type"""
        regret_bounds = {
            'momentum': "O(GDL√T)",
            'mean_reversion': "O(GDLγ²h√T)",
            'breakout': "O(GD√T log T)",
            'trend_following': "O(GDL√T)",
            'volatility': "O(GD√T)",
            'default': "O(GD√T)"
        }
        
        return regret_bounds.get(strategy_type, regret_bounds['default'])
    
    def select_strategy(self, strategy_id):
        """Select a strategy for the overlay"""
        # Get strategy details
        strategy = self.strategy_ontology.find_strategy_by_id(strategy_id)
        if not strategy:
            return False
            
        # Update control state
        self.control_state['selected_strategy'] = strategy_id
        
        # Reset metrics
        self.metrics = {
            'regret_curve': [],
            'benchmark_delta': [],
            'last_update': datetime.now().isoformat()
        }
        
        # Update regret bound type based on strategy
        strategy_type = strategy.get('type', '').lower()
        if 'momentum' in strategy_type or 'trend' in strategy_type:
            self.control_state['regret_bound_type'] = 'GPC'
        elif 'mean_reversion' in strategy_type or 'oscillator' in strategy_type:
            self.control_state['regret_bound_type'] = 'GRC'
        
        return True
    
    def set_risk_mode(self, mode):
        """Set risk mode (low, medium, high)"""
        if mode not in ['low', 'medium', 'high']:
            return False
            
        self.control_state['risk_mode'] = mode
        
        # Emit risk mode change glyph
        if self.glyph_router:
            self.glyph_router.emit(
                'RISK_MODE_CHANGED',
                payload={
                    'mode': mode,
                    'strategy_id': self.control_state['selected_strategy'],
                    'previous_mode': self.control_state['risk_mode']
                },
                source='StrategicOverlayControl',
                confidence=1.0
            )
        
        return True
    
    def set_forecast_horizon(self, steps):
        """Set forecast horizon in steps"""
        if not isinstance(steps, int) or steps < 1:
            return False
            
        self.control_state['forecast_horizon'] = steps
        return True
    
    def toggle_external_feeds(self, enabled):
        """Toggle external data feeds"""
        self.control_state['external_feeds_enabled'] = bool(enabled)
        return True
    
    def toggle_latent_factors(self, enabled):
        """Toggle latent factors for signal processing"""
        self.control_state['latent_factors_enabled'] = bool(enabled)
        return True
    
    def start_simulation(self):
        """Start live simulation"""
        if self.control_state['selected_strategy'] is None:
            return False
            
        self.control_state['simulation_active'] = True
        
        # Initialize metrics for new simulation
        self.metrics = {
            'regret_curve': [{
                'step': 0,
                'regret': 0,
                'timestamp': datetime.now().isoformat()
            }],
            'benchmark_delta': [{
                'step': 0,
                'delta': 0,
                'timestamp': datetime.now().isoformat()
            }],
            'last_update': datetime.now().isoformat()
        }
        
        # Emit simulation start glyph
        if self.glyph_router:
            self.glyph_router.emit(
                'SIMULATION_STARTED',
                payload={
                    'strategy_id': self.control_state['selected_strategy'],
                    'risk_mode': self.control_state['risk_mode'],
                    'forecast_horizon': self.control_state['forecast_horizon'],
                    'regret_bound_type': self.control_state['regret_bound_type'],
                    'external_feeds': self.control_state['external_feeds_enabled'],
                    'latent_factors': self.control_state['latent_factors_enabled']
                },
                source='StrategicOverlayControl',
                confidence=1.0
            )
        
        return True
    
    def pause_simulation(self):
        """Pause live simulation"""
        self.control_state['simulation_active'] = False
        
        # Emit simulation paused glyph
        if self.glyph_router:
            self.glyph_router.emit(
                'SIMULATION_PAUSED',
                payload={
                    'strategy_id': self.control_state['selected_strategy'],
                    'current_step': len(self.metrics['regret_curve']) - 1,
                    'current_regret': self.metrics['regret_curve'][-1]['regret'] if self.metrics['regret_curve'] else 0
                },
                source='StrategicOverlayControl',
                confidence=1.0
            )
        
        return True
    
    def _calculate_regret(self, signal_data):
        """Calculate regret for new signal observation"""
        if not self.control_state['simulation_active']:
            return
            
        # Get current step
        current_step = len(self.metrics['regret_curve'])
        
        # Calculate regret based on control type
        regret_increment = 0
        
        if self.control_state['regret_bound_type'] == 'GPC':
            # Gradient Perturbation Controller model

            signal_strength = signal_data.get('strength', signal_data.get('initial_strength', 0.5))
            regret_increment = math.sqrt(current_step) * signal_strength * 0.1
        else:
            # Gradient Response Controller model
            # Simplified model: regret grows with log(t) * signal_strength
            signal_strength = signal_data.get('strength', signal_data.get('initial_strength', 0.5))
            regret_increment = math.log(current_step + 2) * signal_strength * 0.2
        
        # Calculate benchmark delta
        # Simplified: random walk with drift based on signal strength
        benchmark_delta = (random.random() - 0.5) * signal_strength * 0.3
        
        # Add to metrics
        current_regret = self.metrics['regret_curve'][-1]['regret'] + regret_increment
        now = datetime.now().isoformat()
        
        self.metrics['regret_curve'].append({
            'step': current_step,
            'regret': current_regret,
            'timestamp': now
        })
        
        self.metrics['benchmark_delta'].append({
            'step': current_step,
            'delta': benchmark_delta,
            'timestamp': now
        })
        
        self.metrics['last_update'] = now
```

### 12.6 Codex GPT Chronicle (LLM Reflection Logging)

The Codex GPT Chronicle implements a sophisticated logging and reflection system for all LLM-generated insights and strategies, enabling continuous improvement through feedback loops.

```python
class CodexGPTChronicle:
    """
    LLM reflection logging system to track, evaluate, and improve LLM interactions
    
    Features:
    - Structured prompt and response logging
    - Performance correlation to LLM suggestions
    - Automatic prompt improvement
    - Periodic LLM self-reflection
    """
    
    def __init__(self, db_client, llm_client):
        self.db = db_client
        self.llm = llm_client
        self.chronicle_collection = 'llm_chronicles'
        self.reflection_collection = 'llm_reflections'
        
        # Initialize collections
        self._ensure_collections()
        
        # Track statistics
        self.stats = {
            'total_interactions': 0,
            'successful_strategies': 0,
            'failed_strategies': 0,
            'prompt_improvements': 0
        }
    
    def _ensure_collections(self):
        """Ensure required collections exist"""
        for collection in [self.chronicle_collection, self.reflection_collection]:
            if not self.db.collection_exists(collection):
                self.db.create_collection(collection)
    
    def log_interaction(self, prompt, response, metadata=None):
        """
        Log LLM interaction
        
        Parameters:
        - prompt: Text sent to LLM
        - response: Response from LLM
        - metadata: Additional context information
        
        Returns:
        - str: Interaction ID
        """
        interaction_id = str(uuid.uuid4())
        
        # Prepare interaction record
        interaction = {
            'interaction_id': interaction_id,
            'timestamp': datetime.now().isoformat(),
            'prompt': prompt,
            'response': response,
            'metadata': metadata or {},
            'evaluated': False,
            'outcome': None,
            'reflection': None
        }
        
        # Add prompt tokens and approx cost if available
        if hasattr(self.llm, 'get_token_count'):
            prompt_tokens = self.llm.get_token_count(prompt)
            response_tokens = self.llm.get_token_count(response)
            
            interaction['metrics'] = {
                'prompt_tokens': prompt_tokens,
                'response_tokens': response_tokens,
                'total_tokens': prompt_tokens + response_tokens,
                'estimated_cost': self._estimate_cost(prompt_tokens, response_tokens)
            }
        
        # Store in database
        self.db.insert(
            collection=self.chronicle_collection,
            document=interaction
        )
        
        # Update stats
        self.stats['total_interactions'] += 1
        
        return interaction_id
    
    def _estimate_cost(self, prompt_tokens, response_tokens):
        """Estimate cost of LLM interaction"""
        # These rates would be updated based on actual model pricing
        prompt_rate = 0.00001  # per token
        response_rate = 0.00003  # per token
        
        return (prompt_tokens * prompt_rate) + (response_tokens * response_rate)
    
    def record_strategy_outcome(self, interaction_id, outcome_data):
        """
        Record the outcome of a strategy suggested by the LLM
        
        Parameters:
        - interaction_id: ID of the original interaction
        - outcome_data: Dict with outcome information
        
        Returns:
        - bool: Success
        """
        # Get the interaction
        interaction = self.db.query(
            collection=self.chronicle_collection,
            query={'interaction_id': interaction_id},
            limit=1
        )
        
        if not interaction:
            return False
            
        interaction = interaction[0]
        
        # Update with outcome
        interaction['evaluated'] = True
        interaction['outcome'] = outcome_data
        interaction['outcome_recorded_at'] = datetime.now().isoformat()
        
        # Calculate success metrics
        success = outcome_data.get('success', False)
        profit_loss = outcome_data.get('profit_loss', 0)
        
        if success:
            self.stats['successful_strategies'] += 1
        else:
            self.stats['failed_strategies'] += 1
        
        # Update in database
        self.db.update(
            collection=self.chronicle_collection,
            query={'interaction_id': interaction_id},
            update={'$set': interaction}
        )
        
        # Schedule reflection if needed
        if self._should_trigger_reflection():
            self._schedule_reflection()
        
        return True
    
    def _should_trigger_reflection(self):
        """Determine if reflection should be triggered"""
        # Trigger reflection after every 20 evaluated interactions
        evaluated_count = self.stats['successful_strategies'] + self.stats['failed_strategies']
        return evaluated_count % 20 == 0 and evaluated_count > 0
    
    def generate_prompt_improvements(self, interaction_id):
        """
        Generate improvements for a prompt based on its outcome
        
        Parameters:
        - interaction_id: ID of the interaction to improve
        
        Returns:
        - dict: Improvement suggestions
        """
        # Get the interaction
        interaction = self.db.query(
            collection=self.chronicle_collection,
            query={'interaction_id': interaction_id},
            limit=1
        )
        
        if not interaction or not interaction[0]['evaluated']:
            return None
            
        interaction = interaction[0]
        
        # Create reflection prompt
        reflection_prompt = f"""
        # Prompt Improvement Analysis
        
        ## Original Prompt
        {interaction['prompt']}
        
        ## LLM Response
        {interaction['response']}
        
        ## Outcome
        Success: {interaction['outcome'].get('success', False)}
        Profit/Loss: {interaction['outcome'].get('profit_loss', 0)}
        
        ## Additional Context
        {json.dumps(interaction['outcome'].get('context', {}), indent=2)}
        
        Analyze the above prompt that was given to an LLM for trading strategy generation.
        Consider the outcome of applying the strategy.
        
        Provide specific suggestions to improve the prompt, including:
        1. What important context was missing?
        2. How could constraints be better defined?
        3. What assumptions led to problems?
        4. How could the prompt be restructured for better outcomes?
        
        Format your response as a structured improvement plan with concrete examples.
        """
        
        # Get LLM reflection
        try:
            reflection_response = self.llm.generate(reflection_prompt)
            
            # Parse improvements
            improvements = self._parse_improvements(reflection_response)
            
            # Update interaction with reflection
            interaction['reflection'] = {
                'raw_response': reflection_response,
                'improvements': improvements,
                'generated_at': datetime.now().isoformat()
            }
            
            self.db.update(
                collection=self.chronicle_collection,
                query={'interaction_id': interaction_id},
                update={'$set': interaction}
            )
            
            # Update stats
            self.stats['prompt_improvements'] += 1
            
            return improvements
            
        except Exception as e:
            logging.error(f"Failed to generate prompt improvements: {e}")
            return None
    
    def _parse_improvements(self, reflection_text):
        """Parse improvement suggestions from LLM reflection"""
        # Simple parsing for demonstration purposes
        # In production, this would use more sophisticated parsing
        
        improvements = {
            'missing_context': [],
            'constraint_improvements': [],
            'assumption_corrections': [],
            'structural_changes': []
        }
        
        # Extract sections
        sections = {
            'missing_context': r'Missing Context:(.*?)(?:\n\n|\Z)',
            'constraint_improvements': r'Constraint Improvements:(.*?)(?:\n\n|\Z)',
            'assumption_corrections': r'Assumption Corrections:(.*?)(?:\n\n|\Z)',
            'structural_changes': r'Structural Changes:(.*?)(?:\n\n|\Z)'
        }
        
        for key, pattern in sections.items():
            match = re.search(pattern, reflection_text, re.DOTALL)
            if match:
                section_text = match.group(1).strip()
                items = re.findall(r'\n- (.*?)(?:\n|$)', section_text)
                if items:
                    improvements[key] = items
        
        return improvements
    
    def periodic_reflection(self):
        """
        Perform periodic reflection on all interactions
        
        Returns:
        - dict: Reflection results
        """
        # Get recent evaluated interactions
        recent_interactions = self.db.query(
            collection=self.chronicle_collection,
            query={'evaluated': True},
            sort=[('timestamp', -1)],
            limit=50
        )
        
        if not recent_interactions:
            return None
        
        # Prepare data for reflection
        successful = [i for i in recent_interactions if i['outcome'].get('success', False)]
        failed = [i for i in recent_interactions if not i['outcome'].get('success', False)]
        
        success_rate = len(successful) / len(recent_interactions) if recent_interactions else 0
        
        # Generate summary stats
        stats_summary = {
            'total_evaluated': len(recent_interactions),
            'successful': len(successful),
            'failed': len(failed),
            'success_rate': success_rate,
            'average_profit_loss': sum(i['outcome'].get('profit_loss', 0) for i in recent_interactions) / len(recent_interactions) if recent_interactions else 0
        }
        
        # Create reflection prompt
        reflection_prompt = f"""
        # Strategic Trading System Reflection
        
        ## Performance Summary
        Total Strategies Evaluated: {stats_summary['total_evaluated']}
        Success Rate: {stats_summary['success_rate']:.2f}
        Average Profit/Loss: {stats_summary['average_profit_loss']:.4f}
        
        ## Successful Strategy Examples
        {self._format_examples(successful[:3])}
        
        ## Failed Strategy Examples
        {self._format_examples(failed[:3])}
        
        Based on the above data, please reflect on:
        
        1. What patterns distinguish successful from unsuccessful strategies?
        2. What types of market conditions are our prompts failing to address?
        3. What consistent weaknesses exist in our strategy generation?
        4. How can we systematically improve our prompting approach?
        5. What biases or blind spots are evident in our strategy selection?
        
        Provide specific, actionable insights that can be used to improve our trading system.
        """
        
        # Get LLM reflection
        try:
            reflection_response = self.llm.generate(reflection_prompt)
            
            # Store reflection
            reflection_id = str(uuid.uuid4())
            
            reflection = {
                'reflection_id': reflection_id,
                'timestamp': datetime.now().isoformat(),
                'prompt': reflection_prompt,
                'response': reflection_response,
                'stats_summary': stats_summary,
                'interaction_count': len(recent_interactions),
                'interaction_ids': [i['interaction_id'] for i in recent_interactions]
            }
            
            self.db.insert(
                collection=self.reflection_collection,
                document=reflection
            )
            
            # Extract insights
            insights = self._extract_reflection_insights(reflection_response)
            reflection['insights'] = insights
            
            self.db.update(
                collection=self.reflection_collection,
                query={'reflection_id': reflection_id},
                update={'$set': reflection}
            )
            
            return {
                'reflection_id': reflection_id,
                'insights': insights,
                'stats_summary': stats_summary
            }
            
        except Exception as e:
            logging.error(f"Failed to generate system reflection: {e}")
            return None
    
    def _format_examples(self, interactions, max_length=500):
        """Format interaction examples for reflection"""
        if not interactions:
            return "No examples available."
            
        examples = []
        
        for i, interaction in enumerate(interactions):
            # Truncate prompt and response if too long
            prompt = interaction['prompt']
            if len(prompt) > max_length:
                prompt = prompt[:max_length] + "..."
                
            response = interaction['response']
            if len(response) > max_length:
                response = response[:max_length] + "..."
            
            example = f"""
            Example {i+1}:
            - Prompt: {prompt}
            - Strategy Generated: {self._extract_strategy_name(response)}
            - Outcome: {'Success' if interaction['outcome'].get('success', False) else 'Failure'}
            - Profit/Loss: {interaction['outcome'].get('profit_loss', 0):.4f}
            - Market Condition: {interaction['outcome'].get('context', {}).get('market_condition', 'Unknown')}
            """
            
            examples.append(example)
        
        return "\n".join(examples)
    
    def _extract_strategy_name(self, response):
        """Extract strategy name from response"""
        # Simple extraction for demonstration
        match = re.search(r'Strategy: (.*?)(?:\n|$)', response)
        return match.group(1) if match else "Unknown Strategy"
    
    def _extract_reflection_insights(self, reflection_text):

        insights = {
            'patterns': [],
            'market_conditions': [],
            'weaknesses': [],
            'improvements': [],
            'biases': []
        }
        
        # Extract sections
        sections = {
            'patterns': r'Patterns:(.*?)(?:\n\n|\Z)',
            'market_conditions': r'Market Conditions:(.*?)(?:\n\n|\Z)',
            'weaknesses': r'Weaknesses:(.*?)(?:\n\n|\Z)',
            'improvements': r'Improvements:(.*?)(?:\n\n|\Z)',
            'biases': r'Biases:(.*?)(?:\n\n|\Z)'
        }
        
        for key, pattern in sections.items():
            match = re.search(pattern, reflection_text, re.DOTALL)
            if match:
                section_text = match.group(1).strip()
                items = re.findall(r'\n- (.*?)(?:\n|$)', section_text)
                if items:
                    insights[key] = items
        
        return insights
    
    def _schedule_reflection(self):

            """)
        
        return "\n".join(formatted)
    
    def _format_reflection_insights(self, insights):
        """Format reflection insights for meta-prompt"""
        if not insights:
            return "No insights available."
            
        formatted = []
        
        for category, items in insights.items():
            if items:
                formatted.append(f"### {category.title()}")
                for item in items:
                    formatted.append(f"- {item}")
                formatted.append("")
        
        return "\n".join(formatted)
```

### 12.7 Signal Decay Tracker

The Signal Decay Tracker implements time-based decay of trading signals, ensuring that stale signals are properly weighted or discarded.

```python
class SignalDecayTracker:
    """
    Track signal decay over time to prioritize recent signals and discard stale ones
    
    Features:
    - Time-based signal decay
    - Signal prioritization
    - Automated dismissal of stale signals
    - Visualized signal age
    """
    
    def __init__(self, glyph_router=None):
        self.signals = {}
        self.glyph_router = glyph_router
        self.decay_rates = {
            'price_breakout': 0.05,  # Fast decay
            'trend_change': 0.02,    # Medium decay
            'pattern_completion': 0.01,  # Slow decay
            'default': 0.03          # Default decay rate
        }
    
    def register_signal(self, signal_data):
        """
        Register a new signal for tracking
        
        Parameters:
        - signal_data: Dict containing signal information
        
        Returns:
        - str: Signal ID
        """
        # Ensure required fields
        required_fields = ['type', 'symbol', 'timeframe']
        for field in required_fields:
            if field not in signal_data:
                raise ValueError(f"Missing required field: {field}")
        
        # Generate ID if not provided
        signal_id = signal_data.get('id', str(uuid.uuid4()))
        
        # Set signal properties
        signal = {
            'id': signal_id,
            'type': signal_data['type'],
            'symbol': signal_data['symbol'],
            'timeframe': signal_data['timeframe'],
            'created_at': datetime.now(),
            'initial_strength': signal_data.get('strength', 1.0),
            'current_strength': signal_data.get('strength', 1.0),
            'decay_rate': self._get_decay_rate(signal_data['type']),
            'half_life': signal_data.get('half_life', None),  # Optional custom half-life
            'valid_until': self._calculate_valid_until(signal_data),
            'dismissed': False,
            'acted_upon': False,
            'data': signal_data
        }
        
        # Store signal
        self.signals[signal_id] = signal
        
        # Emit signal registered glyph if router available
        if self.glyph_router:
            self.glyph_router.emit(
                'SIGNAL_REGISTERED',
                payload={
                    'signal_id': signal_id,
                    'type': signal['type'],
                    'symbol': signal['symbol'],
                    'timeframe': signal['timeframe'],
                    'initial_strength': signal['initial_strength'],
                    'valid_until': signal['valid_until'].isoformat()
                },
                source='SignalDecayTracker',
                confidence=signal['initial_strength']
            )
        
        return signal_id
    
    def _get_decay_rate(self, signal_type):
        """Get appropriate decay rate for signal type"""
        # Look for exact match
        if signal_type in self.decay_rates:
            return self.decay_rates[signal_type]
            
        # Look for partial match
        for key, rate in self.decay_rates.items():
            if key in signal_type:
                return rate
                
        # Default rate
        return self.decay_rates['default']
    
    def _calculate_valid_until(self, signal_data):
        """Calculate when signal becomes invalid"""
        now = datetime.now()
        
        # Check for explicit expiry
        if 'valid_until' in signal_data:
            if isinstance(signal_data['valid_until'], str):
                try:
                    return datetime.fromisoformat(signal_data['valid_until'])
                except ValueError:
                    pass
            elif isinstance(signal_data['valid_until'], datetime):
                return signal_data['valid_until']
        
        # Calculate based on timeframe
        timeframe = signal_data.get('timeframe', '1h')
        
        # Parse timeframe (e.g., '1h', '4h', '1d')
        if timeframe.endswith('m'):
            minutes = int(timeframe[:-1])
            return now + timedelta(minutes=minutes * 3)  # 3x timeframe
        elif timeframe.endswith('h'):
            hours = int(timeframe[:-1])
            return now + timedelta(hours=hours * 3)  # 3x timeframe
        elif timeframe.endswith('d'):
            days = int(timeframe[:-1])
            return now + timedelta(days=days * 2)  # 2x timeframe
        elif timeframe.endswith('w'):
            weeks = int(timeframe[:-1])
            return now + timedelta(weeks=weeks)
        else:
            # Default to 1 day if unknown format
            return now + timedelta(days=1)
    
    def update_signal_strengths(self):
        """Update all signal strengths based on decay rates"""
        now = datetime.now()
        for signal_id, signal in list(self.signals.items()):
            if signal['dismissed'] or signal['acted_upon']:
                continue
                
            # Calculate time elapsed
            elapsed = (now - signal['created_at']).total_seconds()
            
            # Calculate decay factor
            if signal['half_life'] is not None:
                # Use half-life if provided
                half_life_seconds = signal['half_life'] * 3600  # Convert hours to seconds
                decay_factor = 2 ** (-elapsed / half_life_seconds)
            else:
                # Use exponential decay
                decay_factor = math.exp(-signal['decay_rate'] * elapsed / 3600)  # per hour
            
            # Update current strength
            signal['current_strength'] = signal['initial_strength'] * decay_factor
            
            # Check if signal is now invalid
            if now >= signal['valid_until'] or signal['current_strength'] < 0.1:
                signal['dismissed'] = True
                
                # Emit signal expired glyph if router available
                if self.glyph_router:
                    self.glyph_router.emit(
                        'SIGNAL_EXPIRED',
                        payload={
                            'signal_id': signal_id,
                            'type': signal['type'],
                            'symbol': signal['symbol'],
                            'timeframe': signal['timeframe'],
                            'final_strength': signal['current_strength'],
                            'duration_hours': elapsed / 3600
                        },
                        source='SignalDecayTracker',
                        confidence=0.1
                    )
    
    def get_active_signals(self, min_strength=0.2):
        """Get all active signals above minimum strength"""
        # Ensure strengths are updated
        self.update_signal_strengths()
        
        # Filter active signals
        active_signals = [
            signal for signal_id, signal in self.signals.items()
            if not signal['dismissed'] and not signal['acted_upon'] and signal['current_strength'] >= min_strength
        ]
        
        # Sort by strength (descending)
        active_signals.sort(key=lambda s: s['current_strength'], reverse=True)
        
        return active_signals
    
    def get_signal(self, signal_id):
        """Get a specific signal by ID"""
        signal = self.signals.get(signal_id)
        
        if signal:
            # Update strength before returning
            self.update_signal_strengths()
            
        return signal
    
    def mark_signal_acted_upon(self, signal_id):
        """Mark a signal as acted upon"""
        if signal_id not in self.signals:
            return False
            
        self.signals[signal_id]['acted_upon'] = True
        self.signals[signal_id]['acted_at'] = datetime.now()
        
        # Emit signal acted upon glyph if router available
        if self.glyph_router:
            signal = self.signals[signal_id]
            self.glyph_router.emit(
                'SIGNAL_ACTED_UPON',
                payload={
                    'signal_id': signal_id,
                    'type': signal['type'],
                    'symbol': signal['symbol'],
                    'timeframe': signal['timeframe'],
                    'strength_at_action': signal['current_strength'],
                    'time_to_action_hours': (datetime.now() - signal['created_at']).total_seconds() / 3600
                },
                source='SignalDecayTracker',
                confidence=signal['current_strength']
            )
        
        return True
    
    def dismiss_signal(self, signal_id, reason=None):
        """Manually dismiss a signal"""
        if signal_id not in self.signals:
            return False
            
        self.signals[signal_id]['dismissed'] = True
        self.signals[signal_id]['dismissed_at'] = datetime.now()
        self.signals[signal_id]['dismiss_reason'] = reason
        
        # Emit signal dismissed glyph if router available
        if self.glyph_router:
            signal = self.signals[signal_id]
            self.glyph_router.emit(
                'SIGNAL_DISMISSED',
                payload={
                    'signal_id': signal_id,
                    'type': signal['type'],
                    'symbol': signal['symbol'],
                    'timeframe': signal['timeframe'],
                    'strength_at_dismissal': signal['current_strength'],
                    'time_to_dismissal_hours': (datetime.now() - signal['created_at']).total_seconds() / 3600,
                    'reason': reason
                },
                source='SignalDecayTracker',
                confidence=signal['current_strength']
            )
        
        return True
    
    def get_signal_metrics(self):
        """Get metrics about signal tracking"""
        all_signals = list(self.signals.values())
        active_signals = [s for s in all_signals if not s['dismissed'] and not s['acted_upon']]
        dismissed_signals = [s for s in all_signals if s['dismissed']]
        acted_upon_signals = [s for s in all_signals if s['acted_upon']]
        
        metrics = {
            'total_signals': len(all_signals),
            'active_signals': len(active_signals),
            'dismissed_signals': len(dismissed_signals),
            'acted_upon_signals': len(acted_upon_signals),
            'average_signal_strength': sum(s['current_strength'] for s in active_signals) / len(active_signals) if active_signals else 0,
            'signals_by_type': self._count_by_type(all_signals),
            'signals_by_symbol': self._count_by_symbol(all_signals),
            'action_rate': len(acted_upon_signals) / (len(acted_upon_signals) + len(dismissed_signals)) if (len(acted_upon_signals) + len(dismissed_signals)) > 0 else 0
        }
        
        return metrics
    
    def _count_by_type(self, signals):
        """Count signals by type"""
        counts = {}
        for signal in signals:
            signal_type = signal['type']
            counts[signal_type] = counts.get(signal_type, 0) + 1
        return counts
    
    def _count_by_symbol(self, signals):
        """Count signals by symbol"""
        counts = {}
        for signal in signals:
            symbol = signal['symbol']
            counts[symbol] = counts.get(symbol, 0) + 1
        return counts
    
    def get_signal_decay_visualization_data(self, signal_id=None):
        """Get data for visualizing signal decay"""
        if signal_id:
            # Get specific signal
            signal = self.get_signal(signal_id)
            if not signal:
                return None
                
            signals = [signal]
        else:
            # Get all non-dismissed signals
            signals = [s for s in self.signals.values() if not s['dismissed']]
            
        # Generate projection data
        now = datetime.now()
        projection_hours = 24  # Project 24 hours into future
        
        visualization_data =    - GUI-based risk grid visualization 
    - Risk rebalancing recommendations
    - Visual mapping of deviations from optimal allocation
    """
    
    def __init__(self, risk_manager):
        self.risk_manager = risk_manager
        self.risk_data = {
            'total': 0,
            'maxAllowed': 0.15,  # Default 15% max risk
            'strategies': []
        }
        self.optimization_history = []
        self.rebalance_recommendations = []
    
    def update_risk_data(self):
        """Update risk allocation data from risk manager"""
        # Get current risk exposure
        exposure = self.risk_manager.get_total_risk_exposure()
        
        # Get per-strategy allocation
        strategy_allocations = self.risk_manager.get_strategy_allocations()
        
        # Update risk data
        self.risk_data = {
            'total': exposure['total_risk_percentage'],
            'maxAllowed': exposure['max_risk_percentage'],
            'available': exposure['remaining_risk_percentage'],
            'strategies': strategy_allocations,
            'updated_at': datetime.now().isoformat()
        }
        
        # Generate recommendations if needed
        if self.should_rebalance():
            self.generate_rebalance_recommendations()
        
        return self.risk_data
    
    def should_rebalance(self):
        """Determine if risk rebalance is needed"""
        # Check if we're close to max risk
        if self.risk_data['total'] > self.risk_data['maxAllowed'] * 0.9:
            return True
            
        # Check if risk is heavily concentrated in one strategy
        strategies = self.risk_data['strategies']
        if strategies and len(strategies) > 1:
            max_strategy_risk = max(s['risk'] for s in strategies)
            if max_strategy_risk > self.risk_data['total'] * 0.7:
                return True
                
        # Check for poorly performing strategies holding too much risk
        for strategy in strategies:
            if strategy.get('pnl', 0) < -0.02 and strategy['risk'] > self.risk_data['total'] * 0.3:
                return True
                
        return False
    
    def generate_rebalance_recommendations(self):
        """Generate risk rebalancing recommendations"""
        strategies = self.risk_data['strategies']
        
        # Reset recommendations
        self.rebalance_recommendations = []
        
        # Calculate performance-based optimal allocation
        try:
            # Sort by risk-adjusted return (simplified)
            sorted_strategies = sorted(
                strategies,
                key=lambda s: s.get('pnl', 0) / s['risk'] if s['risk'] > 0 else -999,
                reverse=True
            )
            
            # Calculate optimal vs actual allocation
            total_risk = self.risk_data['total']
            max_allowed = self.risk_data['maxAllowed']
            
            for i, strategy in enumerate(sorted_strategies):
                current_allocation = strategy['risk']
                
                # Underperforming strategies should have reduced risk
                if strategy.get('pnl', 0) < 0:
                    optimal_allocation = min(current_allocation * 0.5, total_risk * 0.1)
                    action = 'reduce'
                # Top performer can have increased risk
                elif i == 0:
                    optimal_allocation = min(current_allocation * 1.5, max_allowed * 0.5)
                    action = 'increase'
                # Middle performers maintain or slightly adjust
                else:
                    optimal_allocation = current_allocation
                    action = 'maintain'
                
                # Only recommend if change is significant
                if abs(optimal_allocation - current_allocation) / max_allowed > 0.02:
                    self.rebalance_recommendations.append({
                        'strategy': strategy['name'],
                        'current_risk': current_allocation,
                        'optimal_risk': optimal_allocation,
                        'action': action,
                        'delta': optimal_allocation - current_allocation
                    })
        except Exception as e:
            logging.error(f"Error generating rebalance recommendations: {e}")
            
        return self.rebalance_recommendations
    
    def get_visualization_data(self):
        """Get data formatted for visualization"""
        return {
            'riskData': self.risk_data,
            'recommendations': self.rebalance_recommendations,
            'historyData': self._get_risk_history_data()
        }
    
    def _get_risk_history_data(self):
        """Get historical risk allocation data"""
        # This would pull from a time-series database in production
        # For now, return placeholder historical data
        return []
    
    def apply_recommendation(self, recommendation_id):
        """Apply a specific rebalance recommendation"""
        # Find the recommendation
        recommendation = next(
            (r for r in self.rebalance_recommendations if r['strategy'] == recommendation_id),
            None
        )
        
        if not recommendation:
            return False
            
        # Apply to risk manager
        success = self.risk_manager.update_strategy_allocation(
            recommendation['strategy'],
            recommendation['optimal_risk']
        )
        
        if success:
            # Record the optimization
            self.optimization_history.append({
                'timestamp': datetime.now().isoformat(),
                'recommendation': recommendation,
                'previous_allocation': recommendation['current_risk'],
                'new_allocation': recommendation['optimal_risk']
            })
            
            # Update risk data
            self.update_risk_data()
            
        return success
```

```python
            'losing_trades': loss_count,
            'win_rate': win_rate,
            'profit_factor': profit_factor,
            'expectancy': expectancy,
            'max_drawdown': max_drawdown,
            'sharpe_ratio': sharpe,
            'total_profit_loss': total_profit + total_loss,
            'avg_profit': avg_profit,
            'avg_loss': avg_loss,
            'regime_performance': regime_performance,
            'last_updated': datetime.now().isoformat(),
            'last_trade_id': trades[-1]['trade_id'] if trades else None
        }
        
        # Store or update
        existing = self.db.query(
            collection=self.performance_collection,
            query={'strategy_id': strategy_id},
            limit=1
        )
        
        if existing:
            self.db.update(
                collection=self.performance_collection,
                query={'strategy_id': strategy_id},
                update={'$set': performance}
            )
        else:
            self.db.insert(
                collection=self.performance_collection,
                document=performance
            )
        
        return performance
    
    def _calculate_equity_curve(self, trades):
        """Calculate equity curve from trades"""
        initial_balance = 10000  # Starting with 10k for visualization
        balances = [initial_balance]
        
        # Sort by exit time
        sorted_trades = sorted(trades, key=lambda t: t['exit_time'])
        
        for trade in sorted_trades:
            current_balance = balances[-1]
            new_balance = current_balance + trade['profit_loss']
            balances.append(new_balance)
        
        return balances
    
    def _calculate_max_drawdown(self, balances):
        """Calculate maximum drawdown from balance history"""
        max_dd = 0
        peak = balances[0]
        
        for balance in balances:
            if balance > peak:
                peak = balance
            
            dd = (peak - balance) / peak
            max_dd = max(max_dd, dd)
        
        return max_dd
    
    def _calculate_regime_performance(self, trades):
        """Calculate performance by market regime"""
        regimes = {}
        
        for trade in trades:
            regime = trade.get('market_condition', 'unknown')
            
            if regime not in regimes:
                regimes[regime] = {
                    'trades': 0,
                    'wins': 0,
                    'losses': 0,
                    'profit_loss': 0
                }
            
            regimes[regime]['trades'] += 1
            
            if trade['profit_loss'] > 0:
                regimes[regime]['wins'] += 1
            else:
                regimes[regime]['losses'] += 1
                
            regimes[regime]['profit_loss'] += trade['profit_loss']
        
        # Calculate win rates
        for regime, stats in regimes.items():
            stats['win_rate'] = stats['wins'] / stats['trades'] if stats['trades'] > 0 else 0
        
        return regimes
    
    def _calculate_sharpe_ratio(self, returns, risk_free_rate=0.02/365):
        """Calculate Sharpe ratio from returns"""
        if not returns:
            return 0
            
        returns_array = np.array(returns)
        excess_returns = returns_array - risk_free_rate
        
        if np.std(excess_returns) == 0:
            return 0
            
        return np.mean(excess_returns) / np.std(excess_returns)

        """Get performance metrics for a strategy"""
        result = self.db.query(
            collection=self.performance_collection,
            query={'strategy_id': strategy_id},
            limit=1
        )
        
        return result[0] if result else None
    
    def get_best_strategy_for_regime(self, regime, min_trades=10):
        """
        Find the best performing strategy for a given market regime
        
        Parameters:
        - regime: Market regime to match
        - min_trades: Minimum number of trades required
        
        Returns:
        - dict: Best strategy with performance
        """
        all_performance = self.db.query(
            collection=self.performance_collection,
            query={},
        )
        
        # Filter strategies with enough trades
        qualified = [p for p in all_performance if p['total_trades'] >= min_trades]
        
        # Sort by regime performance
        regime_performers = []
        
        for perf in qualified:
            regime_stats = perf.get('regime_performance', {}).get(regime)
            
            if regime_stats and regime_stats['trades'] >= min_trades / 2:
                regime_performers.append({
                    'strategy_id': perf['strategy_id'],
                    'win_rate': regime_stats['win_rate'],
                    'profit_loss': regime_stats['profit_loss'],
                    'trades': regime_stats['trades'],
                    'overall_performance': perf
                })
        
        # Sort by profit_loss
        regime_performers.sort(key=lambda x: x['profit_loss'], reverse=True)
        
        return regime_performers[0] if regime_performers else None
    
    def get_strategy_recommendations(self, current_regime, top_n=3):
        """
        Get strategy recommendations based on current market regime
        
        Parameters:
        - current_regime: Current market regime
        - top_n: Number of recommendations to return
        
        Returns:
        - list: Recommended strategies with confidence scores
        """
        all_performance = self.db.query(
            collection=self.performance_collection,
            query={},
        )
        
        recommendations = []
        
        for perf in all_performance:
            regime_stats = perf.get('regime_performance', {}).get(current_regime)
            
            if not regime_stats or regime_stats['trades'] < 5:
                continue
                
            confidence = self._calculate_strategy_confidence(perf, regime_stats)
            
            recommendations.append({
                'strategy_id': perf['strategy_id'],
                'confidence': confidence,
                'win_rate': regime_stats['win_rate'],
                'profit_loss': regime_stats['profit_loss'],
                'trades_in_regime': regime_stats['trades'],
                'overall_trades': perf['total_trades']
            })
        
        # Sort by confidence
        recommendations.sort(key=lambda x: x['confidence'], reverse=True)
        
        return recommendations[:top_n]
    
    def _calculate_strategy_confidence(self, overall_perf, regime_stats):
        """Calculate confidence score for a strategy in a specific regime"""
        # Factors affecting confidence:
        # 1. Win rate in this regime
        # 2. Number of trades in this regime (more = higher confidence)
        # 3. Consistency (lower standard deviation = higher confidence)
        # 4. Recency (more recent trades = higher confidence)
        
        win_rate_factor = regime_stats['win_rate']
        
        # Trade count factor (0.5 - 1.0)
        trade_count = regime_stats['trades']
        trade_factor = min(1.0, trade_count / 20)  # Max out at 20 trades
        
        # Consistency factor
        consistency = 0.8  # Placeholder - would calculate from actual trade variance
        
        # Calculate confidence
        confidence = (win_rate_factor * 0.4) + (trade_factor * 0.3) + (consistency * 0.3)
        
        return min(1.0, confidence)
    
    def generate_performance_report(self, strategy_id=None):
        """
        Generate comprehensive performance report
        
        Parameters:
        - strategy_id: Optional ID for specific strategy (None for all)
        
        Returns:
        - dict: Performance report
        """
        if strategy_id:
            performances = [self.get_strategy_performance(strategy_id)]
            if not performances[0]:
                return None
        else:
            performances = self.db.query(
                collection=self.performance_collection,
                query={}
            )
        
        # Overall performance
        total_profit_loss = sum(p['total_profit_loss'] for p in performances)
        total_trades = sum(p['total_trades'] for p in performances)
        winning_trades = sum(p['winning_trades'] for p in performances)
        
        # Sort by performance
        performances.sort(key=lambda p: p['total_profit_loss'], reverse=True)
        
        # Best and worst strategies
        best_strategies = performances[:3] if len(performances) >= 3 else performances
        worst_strategies = performances[-3:] if len(performances) >= 3 else []
        
        # Regime analysis
        regime_analysis = self._analyze_regimes(performances)
        
        report = {
            'timestamp': datetime.now().isoformat(),
            'strategy_count': len(performances),
            'total_trades': total_trades,
            'total_profit_loss': total_profit_loss,
            'overall_win_rate': winning_trades / total_trades if total_trades > 0 else 0,
            'best_strategies': [
                {'strategy_id': p['strategy_id'], 'profit_loss': p['total_profit_loss']}
                for p in best_strategies
            ],
            'worst_strategies': [
                {'strategy_id': p['strategy_id'], 'profit_loss': p['total_profit_loss']}
                for p in worst_strategies
            ],
            'regime_analysis': regime_analysis
        }
        
        return report
    
    def _analyze_regimes(self, performances):
        """Analyze strategy performance across different market regimes"""
        regimes = {}
        
        for perf in performances:
            for regime, stats in perf.get('regime_performance', {}).items():
                if regime not in regimes:
                    regimes[regime] = {
                        'total_trades': 0,
                        'total_profit_loss': 0,
                        'strategies': []
                    }
                
                regimes[regime]['total_trades'] += stats['trades']
                regimes[regime]['total_profit_loss'] += stats['profit_loss']
                regimes[regime]['strategies'].append({
                    'strategy_id': perf['strategy_id'],
                    'profit_loss': stats['profit_loss'],
                    'win_rate': stats['win_rate']
                })
        
        # Sort strategies within each regime
        for regime, data in regimes.items():
            data['strategies'].sort(key=lambda s: s['profit_loss'], reverse=True)
            data['top_strategy'] = data['strategies'][0] if data['strategies'] else None
        
        return regimes
```

```python
        for timeframe in self.config['timeframes']:
            ohlcv = self.exchange.fetch_ohlcv(
                self.symbol, 
                timeframe=timeframe, 
                limit=max(100, self.config['min_sample_size'])
            )
            df = pd.DataFrame(
                ohlcv, 
                columns=['timestamp', 'open', 'high', 'low', 'close', 'volume']
            )
            df['timestamp'] = pd.to_datetime(df['timestamp'], unit='ms')
            df.set_index('timestamp', inplace=True)
            
            self.data[timeframe] = df
    
    def calculate_indicators(self):
        """Calculate technical indicators for mean reversion"""
        for timeframe, df in self.data.items():
            if df is None or len(df) < self.config['lookback_periods']:
                continue
            
            # Calculate moving averages
            df['sma'] = df['close'].rolling(window=self.config['lookback_periods']).mean()
            
            # Calculate Bollinger Bands
            df['std'] = df['close'].rolling(window=self.config['lookback_periods']).std()
            df['upper_band'] = df['sma'] + self.config['std_dev_threshold'] * df['std']
            df['lower_band'] = df['sma'] - self.config['std_dev_threshold'] * df['std']
            
            # Calculate Z-score
            df['z_score'] = (df['close'] - df['sma']) / df['std']
            
            # Calculate historical reversion probability
            self._calculate_reversion_probability(df)
    
    def _calculate_reversion_probability(self, df):
        """Calculate historical probability of mean reversion"""
        # Minimum sample size check
        if len(df) < self.config['min_sample_size']:
            return
        
        # Find all instances where z-score exceeded threshold
        high_z = df[df['z_score'] > self.config['z_score_entry']]
        low_z = df[df['z_score'] < -self.config['z_score_entry']]
        
        # Calculate success rate of reversion
        high_z_success = 0
        for idx in high_z.index:
            future_idx = df.index.get_indexer([idx])[0] + self.config['lookback_periods']
            if future_idx < len(df):
                future_z = df.iloc[future_idx]['z_score']
                if future_z < self.config['z_score_exit']:
                    high_z_success += 1
        
        low_z_success = 0
        for idx in low_z.index:
            future_idx = df.index.get_indexer([idx])[0] + self.config['lookback_periods']
            if future_idx < len(df):
                future_z = df.iloc[future_idx]['z_score']
                if future_z > -self.config['z_score_exit']:
                    low_z_success += 1
        
        # Calculate probabilities
        high_z_prob = high_z_success / len(high_z) if len(high_z) > 0 else 0
        low_z_prob = low_z_success / len(low_z) if len(low_z) > 0 else 0
        
        # Store in dataframe
        df['high_z_prob'] = high_z_prob
        df['low_z_prob'] = low_z_prob
    
    def check_for_signals(self):
        """Check for mean reversion trading signals"""
        signals = []
        
        # Calculate weighted z-score across timeframes
        weighted_z_score = 0
        total_weight = 0
        
        for i, timeframe in enumerate(self.config['timeframes']):
            if self.data[timeframe] is None or self.data[timeframe].empty:
                continue
                
            weight = self.config['timeframe_weights'][i]
            df = self.data[timeframe]
            
            current_z = df['z_score'].iloc[-1]
            weighted_z_score += current_z * weight
            total_weight += weight
        
        # Normalize weighted z-score
        if total_weight > 0:
            weighted_z_score /= total_weight
        
        # Check for long signal (negative z-score)
        if weighted_z_score < -self.config['z_score_entry']:
            # Calculate entry/exit prices
            current_price = self.data[self.config['timeframes'][0]]['close'].iloc[-1]
            target_price = self.data[self.config['timeframes'][0]]['sma'].iloc[-1]
            
            # Calculate probabilities from shortest timeframe
            reversion_probability = self.data[self.config['timeframes'][0]].get('low_z_prob', 0.5)
            
            # Generate signal
            signal = {
                'type': 'MEAN_REVERSION_LONG',
                'symbol': self.symbol,
                'entry_price': current_price,
                'target_price': target_price,
                'stop_loss': current_price * 0.97,  # 3% stop loss
                'z_score': weighted_z_score,
                'reversion_probability': reversion_probability,
                'timeframe': 'multi',
                'timestamp': datetime.now().isoformat()
            }
            
            signals.append(signal)
        
        # Check for short signal (positive z-score)
        elif weighted_z_score > self.config['z_score_entry']:
            # Calculate entry/exit prices
            current_price = self.data[self.config['timeframes'][0]]['close'].iloc[-1]
            target_price = self.data[self.config['timeframes'][0]]['sma'].iloc[-1]
            
            # Calculate probabilities from shortest timeframe
            reversion_probability = self.data[self.config['timeframes'][0]].get('high_z_prob', 0.5)
            
            # Generate signal
            signal = {
                'type': 'MEAN_REVERSION_SHORT',
                'symbol': self.symbol,
                'entry_price': current_price,
                'target_price': target_price,
                'stop_loss': current_price * 1.03,  # 3% stop loss
                'z_score': weighted_z_score,
                'reversion_probability': reversion_probability,
                'timeframe': 'multi',
                'timestamp': datetime.now().isoformat()
            }
            
            signals.append(signal)
        
        return signals
    
    def execute_trades(self, signals):
        """Execute trades based on signals"""
        executed_trades = []
        
        # Check if we're at max trades
        if len(self.active_trades) >= self.config['max_trades']:
            return executed_trades
        
        # Process each signal
        for signal in signals:
            # Check probability threshold
            if signal['reversion_probability'] < 0.6:  # Minimum 60% probability
                continue
                
            # Calculate position size
            account_balance = self._get_account_balance()
            risk_amount = account_balance * self.config['risk_per_trade']
            
            # Calculate risk per unit
            if signal['type'] == 'MEAN_REVERSION_LONG':
                risk_per_unit = signal['entry_price'] - signal['stop_loss']
            else:
                risk_per_unit = signal['stop_loss'] - signal['entry_price']
                
            # Calculate position size
            position_size = risk_amount / risk_per_unit
            
            # Execute trade
            try:
                side = 'buy' if signal['type'] == 'MEAN_REVERSION_LONG' else 'sell'
                
                order = self.exchange.create_market_order(
                    symbol=self.symbol,
                    side=side,
                    amount=position_size
                )
                
                # Create trade record
                trade = {
                    'id': str(uuid.uuid4()),
                    'signal': signal,
                    'order': order,
                    'entry_time': datetime.now().isoformat(),
                    'position_size': position_size,
                    'status': 'open'
                }
                
                # Store trade
                self.active_trades.append(trade)
                executed_trades.append(trade)
                
                # Set stop loss and take profit orders
                self._set_exit_orders(trade)
                
            except Exception as e:
                logging.error(f"Trade execution failed: {e}")
        
        return executed_trades
    
    def _set_exit_orders(self, trade):
        """Set stop loss and take profit orders"""
        signal = trade['signal']
        
        try:
            # Set stop loss
            sl_side = 'sell' if signal['type'] == 'MEAN_REVERSION_LONG' else 'buy'
            
            self.exchange.create_order(
                symbol=self.symbol,
                type='stop',
                side=sl_side,
                amount=trade['position_size'],
                price=signal['stop_loss'],
                params={
                    'stopPrice': signal['stop_loss']
                }
            )
            
            # Set take profit
            tp_side = 'sell' if signal['type'] == 'MEAN_REVERSION_LONG' else 'buy'
            
            self.exchange.create_order(
                symbol=self.symbol,
                type='limit',
                side=tp_side,
                amount=trade['position_size'],
                price=signal['target_price']
            )
            
        except Exception as e:
            logging.error(f"Failed to set exit orders: {e}")
    
    def _get_account_balance(self):
        """Get account balance"""
        try:
            balance = self.exchange.fetch_balance()
            return balance['total']['USDT']
        except Exception as e:
            logging.error(f"Failed to fetch balance: {e}")
            return 1000  # Default fallback
    
    def update_active_trades(self):
        """Update status of active trades"""
        for trade in self.active_trades:
            if trade['status'] != 'open':
                continue
                
            # Check if stop loss or take profit was hit
            order_id = trade['order']['id']
            
            try:
                # Get order status
                order = self.exchange.fetch_order(order_id)
                
                # Check for filled status
                if order['status'] == 'closed':
                    # Update trade status
                    trade['status'] = 'closed'
                    trade['exit_time'] = datetime.now().isoformat()
                    trade['profit_loss'] = self._calculate_pnl(trade)
                    
                    # Log trade completion
                    logging.info(f"Trade {trade['id']} closed with P&L: {trade['profit_loss']}")
            except Exception as e:
                logging.error(f"Failed to update trade status: {e}")
    
    def _calculate_pnl(self, trade):
        """Calculate trade profit/loss"""
        signal = trade['signal']
        entry_price = signal['entry_price']
        exit_price = self.data[self.config['timeframes'][0]]['close'].iloc[-1]
        position_size = trade['position_size']
        
        if signal['type'] == 'MEAN_REVERSION_LONG':
            return (exit_price - entry_price) * position_size
        else:
            return (entry_price - exit_price) * position_size
    
    def run(self, interval=3600):
        """Run strategy continuously"""
        while True:
            try:
                # Fetch new data
                self.fetch_historical_data()
                
                # Calculate indicators
                self.calculate_indicators()
                
                # Check for signals
                signals = self.check_for_signals()
                
                # Execute trades if signals found
                if signals:
                    self.execute_trades(signals)
                
                # Update active trades
                self.update_active_trades()
                
                # Sleep until next check
                time.sleep(interval)
                
            except Exception as e:
                logging.error(f"Strategy execution error: {e}")
                time.sleep(interval)
```

```python
    def check_signal(self):
        """Check for trading signals based on configured conditions"""
        # Ensure we have all required data
        for tf in self.config['timeframes']:
            if tf not in self.data or self.data[tf] is None:
                logger.warning(f"Missing data for {tf}, skipping signal check")
                return None
        
        # Current price from shortest timeframe
        shortest_tf = min(self.config['timeframes'])
        price = self.data[shortest_tf]['close'].iloc[-1]
        
        # Check if price is in configured range
        if price < self.config['price_range_min'] or price > self.config['price_range_max']:
            logger.debug(f"Price {price} outside target range {self.config['price_range_min']}-{self.config['price_range_max']}")
            return None

            self.indicators['1h']['rsi'].iloc[-1] >

        )

            logger.debug(f"RSI condition not met: {rsi_1h}, rising: {rsi_rising}")
            return None

            self.indicators['5m']['ao'].iloc[-1] > 0 and 
            self.indicators['5m']['ao'].iloc[-2] <= 0
        )
        
        ao_15m_green = (
            self.indicators['15m']['ao'].iloc[-1] > 0 and 
            self.indicators['15m']['ao'].iloc[-2] <= 0
        )

            logger.debug(f"AO condition not met: 5m: {ao_5m_green}, 15m: {ao_15m_green}")
            return None

        vwap_1h = self.indicators['1h']['vwap'].iloc[-1]
        ma20_1h = self.indicators['1h']['ma20'].iloc[-1]
        
        vwap_condition = vwap_1h < price < ma20_1h
        if not vwap_condition:
            logger.debug(f"VWAP condition not met: {vwap_1h} < {price} < {ma20_1h}")
            return None
        
        # All conditions met - generate signal
        signal = self._generate_signal(price)
        
        logger.info(f"SIGNAL GENERATED: {json.dumps(signal, indent=2)}")
        self.signals.append(signal)
        
        # Execute or alert based on mode
        if self.config['execution_mode'] == 'live':
            self._execute_trade(signal)
        else:
            self._send_alert(signal)
            
        return signal
    
    def _generate_signal(self, price):
        """Generate standardized signal object with all required parameters"""
        # Find recent low for stop loss calculation
        shortest_tf = min(self.config['timeframes'])
        recent_low = self.data[shortest_tf]['low'].iloc[-10:].min()
        
        # Calculate stop loss and take profit levels
        sl = recent_low * (1 - self.config['sl_percentage'])
        tp1 = price * (1 + self.config['tp1_percentage'])
        tp2 = price * (1 + self.config['tp2_percentage'])
        
        # Gather indicator values for the signal
        rsi_1h = self.indicators['1h']['rsi'].iloc[-1]
        vwap_1h = self.indicators['1h']['vwap'].iloc[-1]
        ao_5m = self.indicators['5m']['ao'].iloc[-1]
        ao_15m = self.indicators['15m']['ao'].iloc[-1]
        
        signal = {
            "symbol": self.config['symbol'].replace('/', ''),
            "timestamp": datetime.datetime.now().isoformat(),
            "trigger": "BUY_ZONE_DETECTED",
            "price": float(price),
            "indicators": {
                "rsi_1h": float(rsi_1h),
                "vwap_1h": float(vwap_1h),
                "ao_5m": float(ao_5m),
                "ao_15m": float(ao_15m)
            },
            "execution": {
                "entry": float(price),
                "tp1": float(tp1),
                "tp2": float(tp2),
                "sl": float(sl),
                "risk_percentage": self.config['risk_percentage'],
                "leverage": self.config['leverage']
            }
        }
        
        return signal
    
    def _execute_trade(self, signal):
        """Execute trade based on signal (if in live mode)"""
        if self.config['execution_mode'] != 'live':
            logger.warning("Execution attempted when not in live mode")
            return False
            
        try:
            # Calculate position size based on risk percentage
            account_balance = self._get_futures_balance()
            risk_amount = account_balance * self.config['risk_percentage']
            
            # Calculate actual risk per coin
            price = signal['price']
            sl = signal['execution']['sl']
            risk_per_coin = price - sl
            
            # Calculate position size in coins
            position_size = risk_amount / risk_per_coin
            
            # Apply leverage (be careful here!)
            position_size = position_size * self.config['leverage']
            
            # Place the order
            order = self.exchange.create_market_buy_order(
                self.config['symbol'],
                position_size
            )
            
            # Store trade information
            trade = {
                'signal_id': signal['timestamp'],
                'order_id': order['id'],
                'entry_time': datetime.datetime.now().isoformat(),
                'position_size': position_size,
                'entry_price': price,
                'stop_loss': sl,
                'take_profit1': signal['execution']['tp1'],
                'take_profit2': signal['execution']['tp2'],
                'status': 'open'
            }
            
            self.active_trades.append(trade)
            logger.info(f"Trade executed: {json.dumps(trade, indent=2)}")
            
            # Set stop loss and take profit orders
            self._set_stop_loss_take_profit(trade)
            
            return True
            
        except Exception as e:
            logger.error(f"Error executing trade: {e}")
            return False
    
    def _set_stop_loss_take_profit(self, trade):
        """Place stop loss and take profit orders"""
        try:
            # Place stop loss
            sl_order = self.exchange.create_stop_loss_order(
                self.config['symbol'],
                'sell',
                trade['position_size'],
                trade['stop_loss']
            )
            
            # Place take profit orders - split into two parts
            tp1_size = trade['position_size'] * 0.5  # 50% of position
            tp2_size = trade['position_size'] * 0.5  # 50% of position
            
            tp1_order = self.exchange.create_limit_sell_order(
                self.config['symbol'],
                tp1_size,
                trade['take_profit1']
            )
            
            tp2_order = self.exchange.create_limit_sell_order(
                self.config['symbol'],
                tp2_size,
                trade['take_profit2']
            )
            
            # Update trade with order IDs
            trade['sl_order_id'] = sl_order['id']
            trade['tp1_order_id'] = tp1_order['id']
            trade['tp2_order_id'] = tp2_order['id']
            
            logger.info(f"SL/TP orders placed for trade {trade['order_id']}")
            return True
            
        except Exception as e:
            logger.error(f"Error setting SL/TP: {e}")
            return False
    
    def _send_alert(self, signal):
```

#### 11.1.2 Signal Logic Definition - Precise Parameters

The bot implements the following precise trading logic:

```
IF:
  Current Price ∈ [0.5800, 0.5900]
  AND RSI(1h) < 40 AND RSI(1h, t) > RSI(1h, t-3) // Rising over 3 periods
  AND (AO(5m) crosses above 0 OR AO(15m) crosses above 0)
  AND VWAP(1h) < Current Price < MA20(1h)

THEN:
  Position Size = Account Balance × Risk% ÷ (Entry Price - Stop Loss) × Leverage
  Entry = Current Price (Market) or Limit at specified price
  Stop Loss = Recent Low(10 periods) × 0.98 // 2% below recent low
  Take Profit 1 = Entry × 1.05 // 5% gain
  Take Profit 2 = Entry × 1.10 // 10% gain
  Order Split: 50% to TP1, 50% to TP2
```

```python
def execute_futures_trade(exchange, symbol, direction, size_usd, entry_price=None, 
                         leverage=5, stop_loss=None, take_profit=None):
    """
    Execute a futures trade with precise position sizing and risk management
    
    Parameters:
    - exchange: CCXT exchange instance
    - symbol: Trading pair (e.g., 'PI/USDT')
    - direction: 'long' or 'short'
    - size_usd: Position size in USD
    - entry_price: Limit price (None for market)
    - leverage: Leverage multiplier (e.g., 5 for 5x)
    - stop_loss: Stop loss price
    - take_profit: Take profit price (can be a list for multiple targets)
    
    Returns:
    - dict: Trade results with order IDs and parameters
    """
    
    # 1. Set leverage
    try:
        exchange.set_leverage(leverage, symbol)
        logging.info(f"Leverage set to {leverage}x for {symbol}")
    except Exception as e:
        logging.error(f"Failed to set leverage: {e}")
        return {'success': False, 'error': f"Leverage setting failed: {e}"}
    
    # 2. Calculate position size in contracts
    try:
        market_info = exchange.market(symbol)
        contract_size = market_info.get('contractSize', 1)
        
        # Get mark price for calculations
        ticker = exchange.fetch_ticker(symbol)
        mark_price = ticker['last']
        
        # Convert USD size to contract quantity
        quantity = size_usd / mark_price
        
        # Adjust for contract specifications
        if 'limits' in market_info:
            min_amount = market_info['limits']['amount']['min'] or 0
            max_amount = market_info['limits']['amount']['max'] or float('inf')
            amount_precision = market_info['precision']['amount']
            
            # Ensure within limits
            quantity = max(min_amount, min(quantity, max_amount))
            
            # Round to precision
            quantity = round(quantity, amount_precision) if isinstance(amount_precision, int) else quantity
    except Exception as e:
        logging.error(f"Position size calculation failed: {e}")
        return {'success': False, 'error': f"Size calculation failed: {e}"}
    
    # 3. Place the main order
    order_type = 'limit' if entry_price else 'market'
    side = 'buy' if direction == 'long' else 'sell'
    
    order_params = {
        'symbol': symbol,
        'type': order_type,
        'side': side,
        'amount': quantity
    }
    
    if entry_price:
        order_params['price'] = entry_price
    
    try:
        order = exchange.create_order(**order_params)
        order_id = order['id']
        logging.info(f"Order placed: {order_id}, {side} {quantity} {symbol} at {entry_price or 'market'}")
    except Exception as e:
        logging.error(f"Order placement failed: {e}")
        return {'success': False, 'error': f"Order failed: {e}"}
    
    # 4. Place stop loss order
    sl_order = None
    if stop_loss:
        try:
            sl_side = 'sell' if direction == 'long' else 'buy'
            sl_params = {
                'symbol': symbol,
                'type': 'stop',
                'side': sl_side,
                'amount': quantity,
                'price': stop_loss,
                'params': {
                    'stopPrice': stop_loss
                }
            }
            sl_order = exchange.create_order(**sl_params)
            logging.info(f"Stop loss placed at {stop_loss}")
        except Exception as e:
            logging.error(f"Stop loss placement failed: {e}")
    
    # 5. Place take profit order(s)
    tp_orders = []
    if take_profit:
        # Convert to list if single value
        tp_levels = take_profit if isinstance(take_profit, list) else [take_profit]
        tp_side = 'sell' if direction == 'long' else 'buy'
        
        # Determine how to split the position
        split_size = quantity / len(tp_levels)
        
        for i, tp_price in enumerate(tp_levels):
            try:
                tp_params = {
                    'symbol': symbol,
                    'type': 'limit',
                    'side': tp_side,
                    'amount': split_size,
                    'price': tp_price
                }
                tp_order = exchange.create_order(**tp_params)
                tp_orders.append(tp_order)
                logging.info(f"Take profit #{i+1} placed at {tp_price}")
            except Exception as e:
                logging.error(f"Take profit placement failed for level {i+1}: {e}")
    
    # Return comprehensive trade information
    return {
        'success': True,
        'symbol': symbol,
        'direction': direction,
        'size_usd': size_usd,
        'quantity': quantity,
        'leverage': leverage,
        'entry': {
            'price': entry_price or mark_price,
            'type': order_type,
            'order_id': order_id,
            'order': order
        },
        'stop_loss': {
            'price': stop_loss,
            'order_id': sl_order['id'] if sl_order else None
        },
        'take_profit': [
            {'price': tp_levels[i], 'order_id': tp_orders[i]['id'] if i < len(tp_orders) else None}
            for i in range(len(tp_levels))
        ],
        'timestamp': datetime.now().isoformat()
    }
```

```python
class RiskCalculator:

    def __init__(self, account_balance, max_risk_per_trade=0.03,

        self.max_risk_per_trade = max_risk_per_trade
        self.max_total_risk = max_total_risk
        self.max_leverage = max_leverage
        self.active_positions = []

        """
        Calculate position size based on risk parameters
        
        Parameters:

        - stop_loss: Stop loss price
        - leverage: Planned leverage (capped at max_leverage)
        
        Returns:
        - dict: Position sizing details
        """
        # Calculate actual price risk
        if entry_price <= stop_loss:
            raise ValueError("Entry price must be greater than stop loss for long positions")
        
        price_risk_pct = abs(entry_price - stop_loss) / entry_price
        
        # Apply leverage (if provided)
        effective_leverage = min(leverage or 1, self.max_leverage)
        
        # Calculate maximum risk amount in USDT
        max_risk_amount = self.account_balance * self.max_risk_per_trade
        )
    
    def _match_pattern(self, glyph_name, pattern_regex):
        """Check if glyph matches pattern"""
        return bool(pattern_regex.match(glyph_name))
    
    def _find_matching_subscribers(self, glyph):
        """Find subscribers matching a glyph"""
        matching = []
        
        for pattern, subscriptions in self.subscribers.items():
            # Check if pattern matches
            pattern_regex = subscriptions[0]['pattern_regex']
            if self._match_pattern(glyph, pattern_regex):
                matching.extend(subscriptions)
                
        return matching
    
    def _passes_filters(self, event, filters):
        """Check if event passes subscriber filters"""
        payload = event['payload']
        
        for key, filter_value in filters.items():
            # Handle nested keys with dot notation
            parts = key.split('.')
            value = payload
            
            for part in parts:
                if isinstance(value, dict) and part in value:
                    value = value[part]
                else:
                    # Key not found
                    return False
            
            # Apply filter
            if isinstance(filter_value, dict):
                # Operator-based filter
                for op, op_value in filter_value.items():
                    if op == 'eq' and value != op_value:
                        return False
                    elif op == 'neq' and value == op_value:
                        return False
                    elif op == 'gt' and not (value > op_value):
                        return False
                    elif op == 'lt' and not (value < op_value):
                        return False
                    elif op == 'gte' and not (value >= op_value):
                        return False
                    elif op == 'lte' and not (value <= op_value):
                        return False
                    elif op == 'in' and value not in op_value:
                        return False
                    elif op == 'nin' and value in op_value:
                        return False
                    elif op == 'contains' and not (op_value in value):
                        return False
            else:
                # Simple equality check
                if value != filter_value:
                    return False
                    
        return True
    
    def _log_event(self, event):
        """Log emitted event"""
        self.activity_log.append({
            'event_id': event['id'],
            'glyph': event['glyph'],
            'source': event['source'],

        })
        
        # Trim log if needed

    def get_activity_log(self, limit=None):
```

```python
        """Extract text with regex pattern"""

    def _extract_list(self, text, pattern, default=None):
        """Extract list from text"""
        default = default or []

            return default

        return [item.strip() for item in items]
    
    def _extract_conditions(self, text, section_name):
        """Extract conditions from section"""
        section_pattern = f"{section_name}:(.*?)(?:$|(?:{section_name}))"
        section_match = re.search(section_pattern, text, re.DOTALL)
        
        if not section_match:
            return []
        
        section_text = section_match.group(1)
        condition_lines = [line.strip() for line in section_text.split('\n') if line.strip()]
        
        conditions = []
        for line in condition_lines:
            # Parse condition in format: "indicator operator value"
            parts = re.match(r'([a-zA-Z0-9_]+)\s*([<>=!]+)\s*([0-9.]+)', line)
            if parts:
                indicator, operator, value = parts.groups()
                conditions.append({
                    'indicator': indicator,
                    'operator': operator,
                    'value': float(value) if '.' in value else int(value)
                })
        
        return conditions

            risk_model['stop_loss_pct'] = float(sl_match.group(1)) / 100.0
        
        # Extract take profit
        tp_match = re.search(r'Take Profit:\s*([0-9.]+)%', text)
        if tp_match:
            risk_model['take_profit_pct'] = float(tp_match.group(1)) / 100.0
        
        return risk_model
```

```python
class StrategyPerformanceTracker:
    """

    def _detect_market_condition(self, symbol, timestamp):

    def _update_strategy_performance(self, strategy_id):
        """Update aggregate performance for a strategy"""
        # Get all trades for this strategy
        trades = self.db.query(
            collection=self.trade_collection,
            query={'strategy_id': strategy_id}
        )
        
        if not trades:
            return
            
        # Calculate performance metrics
        total_trades = len(trades)
        winning_trades = [t for t in trades if t['profit_loss'] > 0]
        losing_trades = [t for t in trades if t['profit_loss'] <= 0]
        
        win_count = len(winning_trades)
        loss_count = len(losing_trades)
        
        # Basic metrics
        win_rate = win_count / total_trades if total_trades > 0 else 0
        
        total_profit = sum(t['profit_loss'] for t in winning_trades)
        total_loss = sum(t['profit_loss'] for t in losing_trades)
        
        profit_factor = total_profit / abs(total_loss) if total_loss != 0 else float('inf')
        
        # Calculate average values
        avg_profit = total_profit / win_count if win_count > 0 else 0
        avg_loss = total_loss / loss_count if loss_count > 0 else 0
        
        # Calculate expectancy
        expectancy = (win_rate * avg_profit) - ((1 - win_rate) * abs(avg_loss))
        
        # Calculate drawdown
        balances = self._calculate_equity_curve(trades)
        max_drawdown = self._calculate_max_drawdown(balances)
        
        # Calculate market regime performance
        regime_performance = self._calculate_regime_performance(trades)
        
        # Calculate Sharpe ratio
        returns = [t['return_pct'] for t in trades]
        sharpe = self._calculate_sharpe_ratio(returns)
        
        # Build performance record
        performance = {
            'strategy_id': strategy_id,
            'total_trades': total_trades,
            'winning_trades': win_count,
            'losing_trades': loss_count,
            'win_rate': win_rate,
            'profit_factor': profit_factor,
            'expectancy': expectancy,
            'max_drawdown': max_drawdown,
            'sharpe_ratio': sharpe,
            'total_profit_loss': total_profit + total_loss,
            'avg_profit': avg_profit,
            'avg_loss': avg_loss,
            'regime_performance': regime_performance,
            'last_updated': datetime.now().isoformat(),
            'last_trade_id': trades[-1]['trade_id'] if trades else None
        }
        
        # Store or update
        existing = self.db.query(
            collection=self.performance_collection,
            query={'strategy_id': strategy_id},
            limit=1
        )
        
        if existing:
            self.db.update(
                collection=self.performance_collection,
                query={'strategy_id': strategy_id},
                update={'$set': performance}
            )
        else:
            self.db.insert(
                collection=self.performance_collection,
                document=performance
            )
        
        return performance
    
    def _calculate_equity_curve(self, trades):
        """Calculate equity curve from trades"""
        initial_balance = 10000  # Starting with 10k for visualization
        balances = [initial_balance]
        
        # Sort by exit time
        sorted_trades = sorted(trades, key=lambda t: t['exit_time'])
        
        for trade in sorted_trades:
            current_balance = balances[-1]
            new_balance = current_balance + trade['profit_loss']
            balances.append(new_balance)
        
        return balances
    
    def _calculate_max_drawdown(self, balances):
        """Calculate maximum drawdown from balance history"""
        max_dd = 0
        peak = balances[0]
        
        for balance in balances:
            if balance > peak:
                peak = balance
            
            dd = (peak - balance) / peak
            max_dd = max(max_dd, dd)
        
        return max_dd
    
    def _calculate_regime_performance(self, trades):
        """Calculate performance by market regime"""
        regimes = {}
        
        for trade in trades:
            regime = trade.get('market_condition', 'unknown')
            
            if regime not in regimes:
                regimes[regime] = {
                    'trades': 0,
                    'wins': 0,
                    'losses': 0,
                    'profit_loss': 0
                }
            
            regimes[regime]['trades'] += 1
            
            if trade['profit_loss'] > 0:
                regimes[regime]['wins'] += 1
            else:
                regimes[regime]['losses'] += 1
                
            regimes[regime]['profit_loss'] += trade['profit_loss']
        
        # Calculate win rates
        for regime, stats in regimes.items():
            stats['win_rate'] = stats['wins'] / stats['trades'] if stats['trades'] > 0 else 0
        
        return regimes
    
    def _calculate_sharpe_ratio(self, returns, risk_free_rate=0.02/365):
        """Calculate Sharpe ratio from returns"""
        if not returns:
            return 0
            
        returns_array = np.array(returns)
        excess_returns = returns_array - risk_free_rate
        
        if np.std(excess_returns) == 0:
            return 0
            
        return np.mean(excess_returns) / np.std(excess_returns)

    def _calculate_strategy_confidence(self, overall_perf, regime_stats):
        """Calculate confidence score for a strategy in a specific regime"""
        # Factors affecting confidence:
        # 1. Win rate in this regime
        # 2. Number of trades in this regime (more = higher confidence)
        # 3. Consistency (lower standard deviation = higher confidence)
        # 4. Recency (more recent trades = higher confidence)
        
        win_rate_factor = regime_stats['win_rate']
        
        # Trade count factor (0.5 - 1.0)

        # Consistency factor
        consistency = 0.8  # Placeholder - would calculate from actual trade variance
        
        # Calculate confidence
        confidence = (win_rate_factor * 0.4) + (trade_factor * 0.3) + (consistency * 0.3)
        
        return min(1.0, confidence)

        if strategy_id:
            performances = [self.get_strategy_performance(strategy_id)]
            if not performances[0]:
                return None
        else:
            performances = self.db.query(
                collection=self.performance_collection,
                query={}
            )
        
        # Overall performance
        total_profit_loss = sum(p['total_profit_loss'] for p in performances)
        total_trades = sum(p['total_trades'] for p in performances)
        winning_trades = sum(p['winning_trades'] for p in performances)
        
        # Sort by performance
        performances.sort(key=lambda p: p['total_profit_loss'], reverse=True)
        
        # Best and worst strategies
        best_strategies = performances[:3] if len(performances) >= 3 else performances
        worst_strategies = performances[-3:] if len(performances) >= 3 else []
        
        # Regime analysis
        regime_analysis = self._analyze_regimes(performances)
        
        report = {
            'timestamp': datetime.now().isoformat(),
            'strategy_count': len(performances),
            'total_trades': total_trades,
            'total_profit_loss': total_profit_loss,
            'overall_win_rate': winning_trades / total_trades if total_trades > 0 else 0,
            'best_strategies': [
                {'strategy_id': p['strategy_id'], 'profit_loss': p['total_profit_loss']}
                for p in best_strategies
            ],
            'worst_strategies': [
                {'strategy_id': p['strategy_id'], 'profit_loss': p['total_profit_loss']}
                for p in worst_strategies
            ],
            'regime_analysis': regime_analysis
        }
        
        return report
    
    def _analyze_regimes(self, performances):
        """Analyze strategy performance across different market regimes"""
        regimes = {}
        
        for perf in performances:
            for regime, stats in perf.get('regime_performance', {}).items():
                if regime not in regimes:
                    regimes[regime] = {
                        'total_trades': 0,
                        'total_profit_loss': 0,
                        'strategies': []
                    }
                
                regimes[regime]['total_trades'] += stats['trades']
                regimes[regime]['total_profit_loss'] += stats['profit_loss']
                regimes[regime]['strategies'].append({
                    'strategy_id': perf['strategy_id'],
                    'profit_loss': stats['profit_loss'],
                    'win_rate': stats['win_rate']
                })
        
        # Sort strategies within each regime
        for regime, data in regimes.items():
            data['strategies'].sort(key=lambda s: s['profit_loss'], reverse=True)
            data['top_strategy'] = data['strategies'][0] if data['strategies'] else None
        
        return regimes
```

```python
class StatisticalMeanReversionStrategy:
    """
    Statistical mean reversion strategy with multi-timeframe analysis
    
    Features:
    - Identify overbought/oversold conditions using Bollinger Bands
    - Calculate Z-score based deviation from mean
    - Apply trade probability based on historical reversion analysis
    """
    
    def __init__(self, exchange, symbol='BTC/USDT', config=None):
        self.exchange = exchange
        self.symbol = symbol
        self.config = config or {
            'lookback_periods': 20,
            'std_dev_threshold': 2.0,
            'z_score_entry': 2.0,
            'z_score_exit': 0.0,
            'timeframes': ['1h', '4h', '1d'],
            'timeframe_weights': [0.2, 0.3, 0.5],
            'min_sample_size': 100,
            'risk_per_trade': 0.02,
            'max_trades': 3
        }
        
        self.data = {tf: None for tf in self.config['timeframes']}
        self.signals = []
        self.active_trades = []
    
    def fetch_historical_data(self):
        """Fetch historical data for all timeframes"""
        for timeframe in self.config['timeframes']:
            ohlcv = self.exchange.fetch_ohlcv(
                self.symbol, 
                timeframe=timeframe, 
                limit=max(100, self.config['min_sample_size'])
            )
            df = pd.DataFrame(
                ohlcv, 
                columns=['timestamp', 'open', 'high', 'low', 'close', 'volume']
            )
            df['timestamp'] = pd.to_datetime(df['timestamp'], unit='ms')
            df.set_index('timestamp', inplace=True)
            
            self.data[timeframe] = df
    
    def calculate_indicators(self):
        """Calculate technical indicators for mean reversion"""
        for timeframe, df in self.data.items():
            if df is None or len(df) < self.config['lookback_periods']:
                continue
            
            # Calculate moving averages
            df['sma'] = df['close'].rolling(window=self.config['lookback_periods']).mean()
            
            # Calculate Bollinger Bands
            df['std'] = df['close'].rolling(window=self.config['lookback_periods']).std()
            df['upper_band'] = df['sma'] + self.config['std_dev_threshold'] * df['std']
            df['lower_band'] = df['sma'] - self.config['std_dev_threshold'] * df['std']
            
            # Calculate Z-score
            df['z_score'] = (df['close'] - df['sma']) / df['std']
            
            # Calculate historical reversion probability
            self._calculate_reversion_probability(df)
    
    def _calculate_reversion_probability(self, df):
        """Calculate historical probability of mean reversion"""
        # Minimum sample size check
        if len(df) < self.config['min_sample_size']:
            return
        
        # Find all instances where z-score exceeded threshold
        high_z = df[df['z_score'] > self.config['z_score_entry']]
        low_z = df[df['z_score'] < -self.config['z_score_entry']]
        
        # Calculate success rate of reversion
        high_z_success = 0
        for idx in high_z.index:
            future_idx = df.index.get_indexer([idx])[0] + self.config['lookback_periods']
            if future_idx < len(df):
                future_z = df.iloc[future_idx]['z_score']
                if future_z < self.config['z_score_exit']:
                    high_z_success += 1
        
        low_z_success = 0
        for idx in low_z.index:
            future_idx = df.index.get_indexer([idx])[0] + self.config['lookback_periods']
            if future_idx < len(df):
                future_z = df.iloc[future_idx]['z_score']
                if future_z > -self.config['z_score_exit']:
                    low_z_success += 1
        
        # Calculate probabilities
        high_z_prob = high_z_success / len(high_z) if len(high_z) > 0 else 0
        low_z_prob = low_z_success / len(low_z) if len(low_z) > 0 else 0
        
        # Store in dataframe
        df['high_z_prob'] = high_z_prob
        df['low_z_prob'] = low_z_prob
    
    def check_for_signals(self):
        """Check for mean reversion trading signals"""
        signals = []
        
        # Calculate weighted z-score across timeframes
        weighted_z_score = 0
        total_weight = 0
        
        for i, timeframe in enumerate(self.config['timeframes']):
            if self.data[timeframe] is None or self.data[timeframe].empty:
                continue
                
            weight = self.config['timeframe_weights'][i]
            df = self.data[timeframe]
            
            current_z = df['z_score'].iloc[-1]
            weighted_z_score += current_z * weight
            total_weight += weight
        
        # Normalize weighted z-score
        if total_weight > 0:
            weighted_z_score /= total_weight
        
        # Check for long signal (negative z-score)
        if weighted_z_score < -self.config['z_score_entry']:
            # Calculate entry/exit prices
            current_price = self.data[self.config['timeframes'][0]]['close'].iloc[-1]
            target_price = self.data[self.config['timeframes'][0]]['sma'].iloc[-1]
            
            # Calculate probabilities from shortest timeframe
            reversion_probability = self.data[self.config['timeframes'][0]].get('low_z_prob', 0.5)
            
            # Generate signal
            signal = {
                'type': 'MEAN_REVERSION_LONG',
                'symbol': self.symbol,
                'entry_price': current_price,
                'target_price': target_price,
                'stop_loss': current_price * 0.97,  # 3% stop loss
                'z_score': weighted_z_score,
                'reversion_probability': reversion_probability,
                'timeframe': 'multi',
                'timestamp': datetime.now().isoformat()
            }
            
            signals.append(signal)
        
        # Check for short signal (positive z-score)
        elif weighted_z_score > self.config['z_score_entry']:
            # Calculate entry/exit prices
            current_price = self.data[self.config['timeframes'][0]]['close'].iloc[-1]
            target_price = self.data[self.config['timeframes'][0]]['sma'].iloc[-1]
            
            # Calculate probabilities from shortest timeframe
            reversion_probability = self.data[self.config['timeframes'][0]].get('high_z_prob', 0.5)
            
            # Generate signal
            signal = {
                'type': 'MEAN_REVERSION_SHORT',
                'symbol': self.symbol,
                'entry_price': current_price,
                'target_price': target_price,
                'stop_loss': current_price * 1.03,  # 3% stop loss
                'z_score': weighted_z_score,
                'reversion_probability': reversion_probability,
                'timeframe': 'multi',
                'timestamp': datetime.now().isoformat()
            }
            
            signals.append(signal)
        
        return signals
    
    def execute_trades(self, signals):
        """Execute trades based on signals"""
        executed_trades = []
        
        # Check if we're at max trades
        if len(self.active_trades) >= self.config['max_trades']:
            return executed_trades
        
        # Process each signal
        for signal in signals:
            # Check probability threshold
            if signal['reversion_probability'] < 0.6:  # Minimum 60% probability
                continue
                
            # Calculate position size
            account_balance = self._get_account_balance()
            risk_amount = account_balance * self.config['risk_per_trade']
            
            # Calculate risk per unit
            if signal['type'] == 'MEAN_REVERSION_LONG':
                risk_per_unit = signal['entry_price'] - signal['stop_loss']
            else:
                risk_per_unit = signal['stop_loss'] - signal['entry_price']
                
            # Calculate position size
            position_size = risk_amount / risk_per_unit
            
            # Execute trade
            try:
                side = 'buy' if signal['type'] == 'MEAN_REVERSION_LONG' else 'sell'
                
                order = self.exchange.create_market_order(
                    symbol=self.symbol,
                    side=side,
                    amount=position_size
                )
                
                # Create trade record
                trade = {
                    'id': str(uuid.uuid4()),
                    'signal': signal,
                    'order': order,
                    'entry_time': datetime.now().isoformat(),
                    'position_size': position_size,
                    'status': 'open'
                }
                
                # Store trade
                self.active_trades.append(trade)
                executed_trades.append(trade)
                
                # Set stop loss and take profit orders
                self._set_exit_orders(trade)
                
            except Exception as e:
                logging.error(f"Trade execution failed: {e}")
        
        return executed_trades
    
    def _set_exit_orders(self, trade):
        """Set stop loss and take profit orders"""
        signal = trade['signal']
        
        try:
            # Set stop loss
            sl_side = 'sell' if signal['type'] == 'MEAN_REVERSION_LONG' else 'buy'
            
            self.exchange.create_order(
                symbol=self.symbol,
                type='stop',
                side=sl_side,
                amount=trade['position_size'],
                price=signal['stop_loss'],
                params={
                    'stopPrice': signal['stop_loss']
                }
            )
            
            # Set take profit
            tp_side = 'sell' if signal['type'] == 'MEAN_REVERSION_LONG' else 'buy'
            
            self.exchange.create_order(
                symbol=self.symbol,
                type='limit',
                side=tp_side,
                amount=trade['position_size'],
                price=signal['target_price']
            )
            
        except Exception as e:
            logging.error(f"Failed to set exit orders: {e}")
    
    def _get_account_balance(self):
        """Get account balance"""
        try:
            balance = self.exchange.fetch_balance()
            return balance['total']['USDT']
        except Exception as e:
            logging.error(f"Failed to fetch balance: {e}")
            return 1000  # Default fallback
    
    def update_active_trades(self):
        """Update status of active trades"""
        for trade in self.active_trades:
            if trade['status'] != 'open':
                continue
                
            # Check if stop loss or take profit was hit
            order_id = trade['order']['id']
            
            try:
                # Get order status
                order = self.exchange.fetch_order(order_id)
                
                # Check for filled status
                if order['status'] == 'closed':
                    # Update trade status
                    trade['status'] = 'closed'
                    trade['exit_time'] = datetime.now().isoformat()
                    trade['profit_loss'] = self._calculate_pnl(trade)
                    
                    # Log trade completion
                    logging.info(f"Trade {trade['id']} closed with P&L: {trade['profit_loss']}")
            except Exception as e:
                logging.error(f"Failed to update trade status: {e}")
    
    def _calculate_pnl(self, trade):
        """Calculate trade profit/loss"""
        signal = trade['signal']
        entry_price = signal['entry_price']
        exit_price = self.data[self.config['timeframes'][0]]['close'].iloc[-1]
        position_size = trade['position_size']
        
        if signal['type'] == 'MEAN_REVERSION_LONG':
            return (exit_price - entry_price) * position_size
        else:
            return (entry_price - exit_price) * position_size
    
    def run(self, interval=3600):
        """Run strategy continuously"""
        while True:
            try:
                # Fetch new data
                self.fetch_historical_data()

                signals = self.check_for_signals()
                
                # Execute trades if signals found
                if signals:
                    self.execute_trades(signals)
                
                # Update active trades
                self.update_active_trades()
                
                # Sleep until next check
                time.sleep(interval)
                
            except Exception as e:
                logging.error(f"Strategy execution error: {e}")
                time.sleep(interval)
```

```python
def execute_trade(self, trade_parameters):
    """Execute a trade based on provided parameters"""
    # Extract parameters
    asset = trade_parameters.get('asset')
    side = trade_parameters.get('side')
    position_size = trade_parameters.get('position_size')
    entry_price = trade_parameters.get('entry_price')
    stop_loss = trade_parameters.get('stop_loss')
    take_profit = trade_parameters.get('take_profit')
    order_type = trade_parameters.get('order_type', 'market')
    
    # Validate parameters
    if not all([asset, side, position_size]):
        return {'success': False, 'error': 'Missing required parameters'}
    
    # Convert position size to actual order quantity
    quantity = self._calculate_order_quantity(asset, position_size, entry_price)
    
    # Create order parameters
    order_params = {
        'symbol': asset,
        'side': side.upper(),
        'type': order_type.upper(),
        'quantity': quantity
    }
    
    # Add price for limit orders
    if order_type.lower() == 'limit' and entry_price:
        order_params['price'] = entry_price
    
    # Execute order
    try:
        # Connect to exchange API and place order
        order_result = self._place_order(order_params)
        
        # Set stop loss and take profit if provided
        if order_result.get('orderId') and (stop_loss or take_profit):
            self._set_stop_loss_take_profit(
                order_result['orderId'], 
                stop_loss, 
                take_profit
            )
        
        # Create trade record
        trade_id = str(uuid.uuid4())
        trade_record = {
            'trade_id': trade_id,
            'timestamp': datetime.now().isoformat(),
            'asset': asset,
            'side': side,
            'quantity': quantity,
            'entry_price': entry_price or order_result.get('price'),
            'stop_loss': stop_loss,
            'take_profit': take_profit,
            'order_type': order_type,
            'order_id': order_result.get('orderId'),
            'status': 'open',
            'signal_type': trade_parameters.get('signal_type'),
            'signal_strength': trade_parameters.get('signal_strength'),
            'params': trade_parameters
        }
        
        # Store trade record
        self._store_trade_record(trade_id, trade_record)
        
        return {'success': True, 'trade_id': trade_id, 'order': order_result}
    except Exception as e:
        return {'success': False, 'error': str(e)}
```

```python
def execute_futures_trade(exchange, symbol, direction, size_usd, entry_price=None, 
                         leverage=5, stop_loss=None, take_profit=None):
    """
    Execute a futures trade with precise position sizing and risk management
    
    Parameters:
    - exchange: CCXT exchange instance
    - symbol: Trading pair (e.g., 'PI/USDT')
    - direction: 'long' or 'short'
    - size_usd: Position size in USD
    - entry_price: Limit price (None for market)
    - leverage: Leverage multiplier (e.g., 5 for 5x)
    - stop_loss: Stop loss price
    - take_profit: Take profit price (can be a list for multiple targets)
    
    Returns:
    - dict: Trade results with order IDs and parameters
    """
    
    # 1. Set leverage
    try:
        exchange.set_leverage(leverage, symbol)
        logging.info(f"Leverage set to {leverage}x for {symbol}")
    except Exception as e:
        logging.error(f"Failed to set leverage: {e}")
        return {'success': False, 'error': f"Leverage setting failed: {e}"}
    
    # 2. Calculate position size in contracts
    try:
        market_info = exchange.market(symbol)
        contract_size = market_info.get('contractSize', 1)
        
        # Get mark price for calculations
        ticker = exchange.fetch_ticker(symbol)
        mark_price = ticker['last']
        
        # Convert USD size to contract quantity
        quantity = size_usd / mark_price
        
        # Adjust for contract specifications
        if 'limits' in market_info:
            min_amount = market_info['limits']['amount']['min'] or 0
            max_amount = market_info['limits']['amount']['max'] or float('inf')
            amount_precision = market_info['precision']['amount']
            
            # Ensure within limits
            quantity = max(min_amount, min(quantity, max_amount))
            
            # Round to precision
            quantity = round(quantity, amount_precision) if isinstance(amount_precision, int) else quantity
    except Exception as e:
        logging.error(f"Position size calculation failed: {e}")
        return {'success': False, 'error': f"Size calculation failed: {e}"}
    
    # 3. Place the main order
    order_type = 'limit' if entry_price else 'market'
    side = 'buy' if direction == 'long' else 'sell'
    
    order_params = {
        'symbol': symbol,
        'type': order_type,
        'side': side,
        'amount': quantity
    }
    
    if entry_price:
        order_params['price'] = entry_price
    
    try:
        order = exchange.create_order(**order_params)
        order_id = order['id']
        logging.info(f"Order placed: {order_id}, {side} {quantity} {symbol} at {entry_price or 'market'}")
    except Exception as e:
        logging.error(f"Order placement failed: {e}")
        return {'success': False, 'error': f"Order failed: {e}"}
    
    # 4. Place stop loss order
    sl_order = None
    if stop_loss:
        try:
            sl_side = 'sell' if direction == 'long' else 'buy'
            sl_params = {
                'symbol': symbol,
                'type': 'stop',
                'side': sl_side,
                'amount': quantity,
                'price': stop_loss,
                'params': {
                    'stopPrice': stop_loss
                }
            }
            sl_order = exchange.create_order(**sl_params)
            logging.info(f"Stop loss placed at {stop_loss}")
        except Exception as e:
            logging.error(f"Stop loss placement failed: {e}")
    
    # 5. Place take profit order(s)
    tp_orders = []
    if take_profit:
        # Convert to list if single value
        tp_levels = take_profit if isinstance(take_profit, list) else [take_profit]
        tp_side = 'sell' if direction == 'long' else 'buy'
        
        # Determine how to split the position
        split_size = quantity / len(tp_levels)
        
        for i, tp_price in enumerate(tp_levels):
            try:
                tp_params = {
                    'symbol': symbol,
                    'type': 'limit',
                    'side': tp_side,
                    'amount': split_size,
                    'price': tp_price
                }
                tp_order = exchange.create_order(**tp_params)
                tp_orders.append(tp_order)
                logging.info(f"Take profit #{i+1} placed at {tp_price}")
            except Exception as e:
                logging.error(f"Take profit placement failed for level {i+1}: {e}")
    
    # Return comprehensive trade information
    return {
        'success': True,
        'symbol': symbol,
        'direction': direction,
        'size_usd': size_usd,
        'quantity': quantity,
        'leverage': leverage,
        'entry': {
            'price': entry_price or mark_price,
            'type': order_type,
            'order_id': order_id,
            'order': order
        },
        'stop_loss': {
            'price': stop_loss,
            'order_id': sl_order['id'] if sl_order else None
        },
        'take_profit': [
            {'price': tp_levels[i], 'order_id': tp_orders[i]['id'] if i < len(tp_orders) else None}
            for i in range(len(tp_levels))
        ],
        'timestamp': datetime.now().isoformat()
    }
```

class RiskCalculator:
    """Trading risk calculator with precise risk management rules"""
    
    def __init__(self, account_balance, max_risk_per_trade=0.03, 
                 max_total_risk=0.15, max_leverage=5):
        """
        Initialize risk calculator
        
        Parameters:
        - account_balance: Total account balance in USDT
        - max_risk_per_trade: Maximum risk percentage per single trade (0.03 = 3%)
        - max_total_risk: Maximum combined risk across all open positions (0.15 = 15%)
        - max_leverage: Maximum allowed leverage (e.g., 5x)
        """
        self.account_balance = account_balance
        self.max_risk_per_trade = max_risk_per_trade
        self.max_total_risk = max_total_risk
        self.max_leverage = max_leverage
        self.active_positions = []
    
    def calculate_position_size(self, entry_price, stop_loss, leverage=None):
        """
        Calculate position size based on risk parameters
        
        Parameters:
        - entry_price: Planned entry price
        - stop_loss: Stop loss price
        - leverage: Planned leverage (capped at max_leverage)
        
        Returns:
        - dict: Position sizing details
        """
        # Calculate actual price risk
        if entry_price <= stop_loss:
            raise ValueError("Entry price must be greater than stop loss for long positions")
        
        price_risk_pct = abs(entry_price - stop_loss) / entry_price
        
        # Apply leverage (if provided)
        effective_leverage = min(leverage or 1, self.max_leverage)
        
        # Calculate maximum risk amount in USDT
        max_risk_amount = self.account_balance * self.max_risk_per_trade
